@OptIn(ExperimentalFoundationApi::class)
class MainActivity : ComponentActivity(), TextToSpeech.OnInitListener {

    private lateinit var tts: TextToSpeech
    private val client = JarvisClient()

    override fun onInit(status: Int) {
        if (status == TextToSpeech.SUCCESS) {
            tts.language = java.util.Locale.forLanguageTag("en-AU")
            tts.setPitch(1.0f)
            tts.setSpeechRate(1.1f)
            val voices = tts.voices ?: return
            val auVoice = voices
                .filter { it.locale.country == "AU" }
                .filter { it.name.contains("local", ignoreCase = true) }
                .minByOrNull { it.latency }
            if (auVoice != null) { tts.voice = auVoice; return }
            val fallback = voices
                .filter { it.name.startsWith("en", ignoreCase = true) }
                .filter { it.name.contains("local", ignoreCase = true) }
                .maxByOrNull { it.quality }
            if (fallback != null) tts.voice = fallback
        }
    }

    private fun speak(text: String) = tts.speak(text, TextToSpeech.QUEUE_FLUSH, null, null)

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        tts = TextToSpeech(this, this)
        enableEdgeToEdge()
        setContent { JarvisOS() }
    }

    override fun onDestroy() { tts.stop(); tts.shutdown(); super.onDestroy() }

    // â”€â”€ Mic launcher lives at class level so it is registered before any composable runs
    private val micLauncher = registerForActivityResult(
        ActivityResultContracts.StartActivityForResult()
    ) { result ->
        Log.d("JARVIS_MIC", "Result received. code=${result.resultCode}")
        val spoken = result.data
            ?.getStringArrayListExtra(SpeechRecognizer.RESULTS_RECOGNITION)
            ?.firstOrNull()
        Log.d("JARVIS_MIC", "Spoken: $spoken")
        if (!spoken.isNullOrBlank()) {
            pendingSpeech = spoken
        }
    }

    // Channel speech result back into the composable world
    private var pendingSpeech: String? = null

    fun launchMic() {
        Log.d("JARVIS_MIC", "launchMic() called from activity")
        val intent = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH).apply {
            putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
            putExtra(RecognizerIntent.EXTRA_PROMPT, "Speak to Jarvis...")
        }
        try {
            micLauncher.launch(intent)
        } catch (e: Exception) {
            Log.e("JARVIS_MIC", "Launch failed: ${e.message}")
        }
    }

    @Composable
    fun JarvisOS() {
        // Dark theme with vivid coloured bubble borders
        val bg        = Color(0xFF080D18)   // deep dark background
        val surface   = Color(0xFF0F1A2E)   // dark card surface
        val border    = Color(0xFF00D9FF)   // bright cyan
        val borderDim = Color(0xFF1E4A5E)   // dim cyan
        val cyan      = Color(0xFF00D9FF)   // bright cyan
        val amber     = Color(0xFFFFBF00)   // bright amber
        val green     = Color(0xFF00F080)   // bright green
        val textPri   = Color(0xFFF0FBFF)   // near white
        val textSec   = Color(0xFF6B8A9A)   // muted blue-grey
        val jarvisCol = Color(0xFF00D9FF)   // cyan for Jarvis label
        val youCol    = Color(0xFFFFBF00)   // amber for You label

        var input        by remember { mutableStateOf("") }
        var isLoading    by remember { mutableStateOf(false) }
        var isMicOn      by remember { mutableStateOf(false) }
        var serverOnline by remember { mutableStateOf(false) }
        var currentTime  by remember { mutableStateOf("") }
        var dashboard    by remember { mutableStateOf<DashboardData?>(null) }
        var latencyMs    by remember { mutableStateOf(0L) }
        var selectedTab     by remember { mutableStateOf(0) }
        // Genesis-053 Sprint-002: sessions awaiting approval
        var pendingSessions by remember { mutableStateOf<List<OrchestratorSession>>(emptyList()) }
        val tabs = listOf("CHAT", "SYSTEM", "MISSION")
        val keyboardController = LocalSoftwareKeyboardController.current

        fun timestamp() = SimpleDateFormat("HH:mm:ss", Locale.getDefault()).format(Date())

        val hour = remember { Calendar.getInstance().get(Calendar.HOUR_OF_DAY) }
        var greeting by remember {
            mutableStateOf(when {
                hour < 12 -> "Good morning. Connecting..."
                hour < 17 -> "Good afternoon. Connecting..."
                hour < 21 -> "Good evening. Connecting..."
                else      -> "Working late. Connecting..."
            })
        }

        data class Message(val text: String, val fromJarvis: Boolean, val timestamp: String)
        val messages = remember { mutableStateListOf(Message(greeting, true, timestamp())) }
        val listState = rememberLazyListState()

        // Send function
        fun send(text: String) {
            if (text.isBlank() || isLoading) return
            val userText = text.trim()
            input = ""
            keyboardController?.hide()
            messages.add(Message(userText, false, timestamp()))
            isLoading = true
            CoroutineScope(Dispatchers.IO).launch {
                val result = client.chat(userText)
                withContext(Dispatchers.Main) {
                    isLoading = false
                    val reply = result.getOrElse { it.message ?: "No response." }
                    messages.add(Message(reply, true, timestamp()))
                    speak(reply)
                }
            }
        }

        // Poll for speech results from the activity-level launcher
        LaunchedEffect(Unit) {
            while (true) {
                delay(100)
                val speech = pendingSpeech
                if (speech != null) {
                    pendingSpeech = null
                    isMicOn = false
                    send(speech)
                }
            }
        }

        LaunchedEffect(messages.size) {
            if (messages.isNotEmpty()) listState.animateScrollToItem(messages.size - 1)
        }

        LaunchedEffect(Unit) {
            while (true) {
                val t0 = System.currentTimeMillis()
                val data = client.dashboard()
                val elapsed = System.currentTimeMillis() - t0
                latencyMs = elapsed
                if (data != null) {
                    dashboard = data.copy(latencyMs = elapsed)
                    serverOnline = true
                    // Genesis-053 Sprint-002: poll for pending approvals in same heartbeat
                    val sessions = client.orchestratorStatus()
                    if (sessions != null) pendingSessions = sessions
                    val tg = when {
                        hour < 12 -> "Good morning."
                        hour < 17 -> "Good afternoon."
                        hour < 21 -> "Good evening."
                        else      -> "Working late."
                    }
                    val newGreeting = "$tg\nAll systems healthy.\nEngineering stable. ${data.testsPassed} tests passing.\n${data.workersOnline} workers online.\nReady when you are."
                    greeting = newGreeting
                    if (messages.size == 1 && messages[0].fromJarvis) {
                        messages[0] = Message(newGreeting, true, timestamp())
                    }
                } else {
                    serverOnline = false
                }
                delay(30_000)
            }
        }

        LaunchedEffect(Unit) {
            while (true) {
                currentTime = SimpleDateFormat("HH:mm", Locale.getDefault()).format(Date())
                delay(30_000)
            }
        }

        val cursorAlpha by rememberInfiniteTransition(label = "cursor").animateFloat(
            initialValue = 1f, targetValue = 0f,
            animationSpec = infiniteRepeatable(tween(500, easing = LinearEasing), RepeatMode.Reverse),
            label = "cursor"
        )

        // File upload launcher
        val fileLauncher = rememberLauncherForActivityResult(
            ActivityResultContracts.GetContent()
        ) { uri: Uri? ->
            if (uri == null) return@rememberLauncherForActivityResult
            CoroutineScope(Dispatchers.IO).launch {
                try {
                    val mime = contentResolver.getType(uri) ?: "application/octet-stream"
                    val fileName = contentResolver
                        .query(uri, null, null, null, null)
                        ?.use { cursor ->
                            val idx = cursor.getColumnIndex(android.provider.OpenableColumns.DISPLAY_NAME)
                            cursor.moveToFirst()
                            if (idx >= 0) cursor.getString(idx) else null
                        } ?: uri.lastPathSegment ?: "upload"
                    val tmpFile = java.io.File(cacheDir, fileName)
                    contentResolver.openInputStream(uri)?.use { input ->
                        tmpFile.outputStream().use { output -> input.copyTo(output) }
                    }
                    withContext(Dispatchers.Main) {
                        messages.add(Message("Uploading: $fileName", false, timestamp()))
                        isLoading = true
                    }
                    val result = client.upload(tmpFile, mime)
                    tmpFile.delete()
                    withContext(Dispatchers.Main) {
                        isLoading = false
                        val reply = result.map { it.message }.getOrElse { it.message ?: "Upload failed." }
                        messages.add(Message(reply, true, timestamp()))
                        speak(reply)
                    }
                } catch (e: Exception) {
                    withContext(Dispatchers.Main) {
                        isLoading = false
                        messages.add(Message("Upload error: ${e.message}", true, timestamp()))
                    }
                }
            }
        }

        fun copyToClipboard(text: String) {
            val clipboard = getSystemService(Context.CLIPBOARD_SERVICE) as ClipboardManager
            clipboard.setPrimaryClip(ClipData.newPlainText("Jarvis", text))
        }

        Box(
            modifier = Modifier
                .fillMaxSize()
                .background(bg)
                .systemBarsPadding()
                .imePadding()
        ) {
            Column(modifier = Modifier.fillMaxSize()) {

                // Header
                Box(
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(horizontal = 12.dp, vertical = 8.dp)
                        .background(surface, RoundedCornerShape(20.dp))
                        .border(1.dp, border.copy(alpha = 0.6f), RoundedCornerShape(20.dp))
                        .padding(horizontal = 16.dp, vertical = 12.dp)
                ) {
                    Column {
                        Text("JARVIS OS", color = cyan, fontFamily = FontFamily.Monospace, fontWeight = FontWeight.Bold, fontSize = 15.sp, letterSpacing = 3.sp)
                        Text("ANDROID ALPHA  v0.1", color = textSec, fontFamily = FontFamily.Monospace, fontSize = 8.sp, letterSpacing = 2.sp)
                    }
                    Row(
                        modifier = Modifier.align(Alignment.CenterEnd),
                        verticalAlignment = Alignment.CenterVertically,
                        horizontalArrangement = Arrangement.spacedBy(8.dp)
                    ) {
                        Text(currentTime, color = textSec, fontFamily = FontFamily.Monospace, fontSize = 11.sp)
                        Row(verticalAlignment = Alignment.CenterVertically, horizontalArrangement = Arrangement.spacedBy(4.dp)) {
                            Box(modifier = Modifier.size(6.dp).background(if (serverOnline) green else amber, RoundedCornerShape(3.dp)))
                            Text(if (serverOnline) "ONLINE" else "OFFLINE", color = if (serverOnline) green else amber, fontFamily = FontFamily.Monospace, fontSize = 8.sp, letterSpacing = 1.sp)
                        }
                    }
                }

                // Tabs
                TabRow(
                    selectedTabIndex = selectedTab,
                    containerColor   = Color.Transparent,
                    contentColor     = cyan,
                    divider          = { Box(Modifier.fillMaxWidth().height(1.dp).background(borderDim)) }
                ) {
                    tabs.forEachIndexed { i, title ->
                        Tab(
                            selected  = selectedTab == i,
                            onClick   = { selectedTab = i },
                            text      = {
                                Text(title, fontFamily = FontFamily.Monospace, fontSize = 10.sp, letterSpacing = 2.sp,
                                    color = if (selectedTab == i) cyan else textSec,
                                    fontWeight = if (selectedTab == i) FontWeight.Bold else FontWeight.Normal)
                            }
                        )
                    }
                }

                if (selectedTab == 1) {
                    SystemDashboard(dashboard, latencyMs, serverOnline, cyan, amber, surface, border, borderDim, textPri, textSec, green)
                }
                if (selectedTab == 2) {
                    MissionControl(
                        data = dashboard, cyan = cyan, amber = amber, green = green,
                        surface = surface, border = border, textPri = textPri,
                        textSec = textSec, bg = bg,
                        pendingSessions = pendingSessions,
                        onApprove = { sessionId ->
                            CoroutineScope(Dispatchers.Main).launch {
                                client.approveSession(sessionId, "approve")
                                val updated = client.orchestratorStatus()
                                if (updated != null) pendingSessions = updated
                            }
                        },
                        onReject = { sessionId ->
                            CoroutineScope(Dispatchers.Main).launch {
                                client.approveSession(sessionId, "reject")
                                val updated = client.orchestratorStatus()
                                if (updated != null) pendingSessions = updated
                            }
                        },
                    )
                }
                if (selectedTab != 0) return@Column

                // Chat messages
                LazyColumn(
                    state = listState,
                    modifier = Modifier.weight(1f).fillMaxWidth().padding(horizontal = 12.dp),
                    verticalArrangement = Arrangement.spacedBy(8.dp),
                    contentPadding = PaddingValues(vertical = 8.dp)
                ) {
                    items(messages) { msg ->
                        MessageCard(msg.text, msg.fromJarvis, msg.timestamp,
                            jarvisCol, youCol, surface, border, borderDim, textPri, textSec,
                            cyanC = cyan, amberC = amber,
                            onLongPress = { copyToClipboard(msg.text) }
                        )
                    }
                    if (isLoading) {
                        item {
                            Box(
                                modifier = Modifier
                                    .fillMaxWidth()
                                    .background(surface, RoundedCornerShape(18.dp))
                                    .border(1.dp, borderDim, RoundedCornerShape(18.dp))
                                    .padding(12.dp)
                            ) {
                                Row(verticalAlignment = Alignment.CenterVertically, horizontalArrangement = Arrangement.spacedBy(6.dp)) {
                                    Text("JARVIS", color = jarvisCol, fontFamily = FontFamily.Monospace, fontWeight = FontWeight.Bold, fontSize = 9.sp, letterSpacing = 2.sp)
                                    Text("Processing", color = textSec, fontFamily = FontFamily.Monospace, fontSize = 12.sp)
                                    Text("\u25cb", color = cyan, fontFamily = FontFamily.Monospace, fontSize = 12.sp, modifier = Modifier.alpha(cursorAlpha))
                                }
                            }
                        }
                    }
                }

                // Input area
                Column(
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(horizontal = 12.dp, vertical = 8.dp)
                        .background(surface, RoundedCornerShape(20.dp))
                        .border(1.dp, border.copy(alpha = 0.5f), RoundedCornerShape(20.dp))
                        .padding(12.dp)
                ) {
                    Text("> INPUT", color = textSec, fontFamily = FontFamily.Monospace, fontSize = 9.sp, letterSpacing = 2.sp, modifier = Modifier.padding(bottom = 6.dp))
                    Row(
                        modifier = Modifier.fillMaxWidth(),
                        verticalAlignment = Alignment.CenterVertically,
                        horizontalArrangement = Arrangement.spacedBy(8.dp)
                    ) {
                        Text("\u25b6", color = cyan, fontFamily = FontFamily.Monospace, fontSize = 14.sp)
                        OutlinedTextField(
                            value = input,
                            onValueChange = { input = it },
                            modifier = Modifier.weight(1f),
                            placeholder = { Text("Enter command...", color = textSec, fontFamily = FontFamily.Monospace, fontSize = 13.sp) },
                            colors = OutlinedTextFieldDefaults.colors(
                                focusedBorderColor      = border,
                                unfocusedBorderColor    = borderDim,
                                cursorColor             = amber,
                                focusedTextColor        = textPri,
                                unfocusedTextColor      = textPri,
                                focusedContainerColor   = Color(0xFF080D18).copy(alpha = 0.5f),
                                unfocusedContainerColor = Color(0xFF080D18).copy(alpha = 0.3f)
                            ),
                            textStyle = TextStyle(fontFamily = FontFamily.Monospace, fontSize = 13.sp, color = textPri),
                            singleLine = false,
                            maxLines = 4,
                            shape = RoundedCornerShape(8.dp)
                        )
                    }
                    Spacer(Modifier.height(10.dp))
                    Row(
                        modifier = Modifier.fillMaxWidth(),
                        horizontalArrangement = Arrangement.SpaceBetween,
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        // MIC â€” plain Box with clickable, no Material wrapper
                        Box(
                            modifier = Modifier
                                .height(40.dp)
                                .widthIn(min = 88.dp)
                                .background(
                                    if (isMicOn) amber.copy(alpha = 0.2f) else Color.Transparent,
                                    RoundedCornerShape(50)
                                )
                                .border(1.dp, if (isMicOn) amber else cyan, RoundedCornerShape(50))
                                .clickable(
                                    interactionSource = remember { MutableInteractionSource() },
                                    indication = null
                                ) {
                                    Log.d("JARVIS_MIC", "MIC box clicked")
                                    isMicOn = true
                                    launchMic()
                                },
                            contentAlignment = Alignment.Center
                        ) {
                            Text(
                                "\u25ce MIC",
                                color = if (isMicOn) amber else cyan,
                                fontFamily = FontFamily.Monospace,
                                fontWeight = FontWeight.Bold,
                                fontSize = 11.sp,
                                letterSpacing = 1.sp
                            )
                        }

                        CreamButton("\u25a0 STOP", Color(0xFF7A1A1A), Color(0xFF7A1A1A).copy(alpha = 0.4f), Color.Transparent, border, borderDim) { tts.stop() }
                        CreamButton("FILE", textSec, borderDim, Color.Transparent, border, borderDim) { fileLauncher.launch("*/*") }
                        CreamButton("SEND \u25b6", if (input.isNotBlank()) surface else textSec, if (input.isNotBlank()) border else borderDim, if (input.isNotBlank()) border.copy(alpha = 0.3f) else Color.Transparent, border, borderDim) { send(input) }
                    }
                }
            }
        }
    }

    @Composable
    fun CreamButton(label: String, color: Color, borderColor: Color, background: Color, activeBorder: Color, dimBorder: Color, onClick: () -> Unit) {
        OutlinedButton(
            onClick = onClick,
            modifier = Modifier.height(40.dp).widthIn(min = 88.dp),
            shape = RoundedCornerShape(50),
            border = androidx.compose.foundation.BorderStroke(1.dp, borderColor),
            colors = ButtonDefaults.outlinedButtonColors(
                containerColor = background,
                contentColor   = color
            ),
            contentPadding = PaddingValues(horizontal = 12.dp, vertical = 0.dp)
        ) {
            Text(label, color = color, fontFamily = FontFamily.Monospace, fontWeight = FontWeight.Bold, fontSize = 11.sp, letterSpacing = 1.sp)
        }
    }

    @OptIn(ExperimentalFoundationApi::class)
    @Composable
    fun MessageCard(
        text: String, fromJarvis: Boolean, timestamp: String,
        jarvisCol: Color, youCol: Color, surface: Color,
        border: Color, borderDim: Color, textPri: Color, textSec: Color,
        cyanC: Color, amberC: Color,
        onLongPress: () -> Unit
    ) {
        var showCopied by remember { mutableStateOf(false) }
        Box(
            modifier = Modifier
                .fillMaxWidth()
                .background(surface, RoundedCornerShape(18.dp))
                .border(1.5.dp, if (fromJarvis) cyanC.copy(alpha = 0.7f) else amberC.copy(alpha = 0.7f), RoundedCornerShape(18.dp))
                .combinedClickable(onClick = {}, onLongClick = { showCopied = true; onLongPress() })
                .padding(horizontal = 14.dp, vertical = 10.dp)
        ) {
            Column {
                Row(modifier = Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.SpaceBetween, verticalAlignment = Alignment.CenterVertically) {
                    Text(if (fromJarvis) "JARVIS" else "YOU", color = if (fromJarvis) jarvisCol else youCol,
                        fontFamily = FontFamily.Monospace, fontWeight = FontWeight.Bold, fontSize = 9.sp, letterSpacing = 3.sp)
                    Row(horizontalArrangement = Arrangement.spacedBy(8.dp), verticalAlignment = Alignment.CenterVertically) {
                        if (showCopied) {
                            LaunchedEffect(Unit) { delay(2000); showCopied = false }
                            Text("COPIED", color = jarvisCol, fontFamily = FontFamily.Monospace, fontSize = 8.sp)
                        }
                        Text(timestamp, color = textSec, fontFamily = FontFamily.Monospace, fontSize = 9.sp)
                    }
                }
                Spacer(Modifier.height(6.dp))
                Text(text, color = textPri, fontFamily = FontFamily.Monospace, fontSize = 13.sp, lineHeight = 20.sp)
            }
        }
    }

    @Composable
    fun SystemDashboard(
        data: DashboardData?, latencyMs: Long, online: Boolean,
        cyan: Color, amber: Color, surface: Color, border: Color, borderDim: Color,
        textPri: Color, textSec: Color, green: Color
    ) {
        val d = data ?: DashboardData()
        LazyColumn(modifier = Modifier.fillMaxSize().padding(12.dp), verticalArrangement = Arrangement.spacedBy(6.dp)) {
            item { DashRow("STATUS",     if (online) "ONLINE" else "OFFLINE", if (online) green else amber, textSec, surface, border, borderDim) }
            item { DashRow("VERSION",    d.version, cyan, textSec, surface, border, borderDim) }
            item { DashRow("UPTIME",     d.uptime, textPri, textSec, surface, border, borderDim) }
            item { DashRow("LATENCY",    "${latencyMs}ms", if (latencyMs < 300) green else amber, textSec, surface, border, borderDim) }
            item { DashRow("WORKERS",    "${d.workersOnline} online", textPri, textSec, surface, border, borderDim) }
            item { DashRow("MEMORY",     "${d.memoryRecords} records / ${d.memoryStatus}", textPri, textSec, surface, border, borderDim) }
            item { DashRow("AI",         d.aiProvider, textPri, textSec, surface, border, borderDim) }
            item { DashRow("GENESIS",    d.currentGenesis, cyan, textSec, surface, border, borderDim) }
            item { DashRow("LAST BUILD", d.lastGenesis, textPri, textSec, surface, border, borderDim) }
            item { DashRow("TESTS",      "${d.testsPassed} passed", green, textSec, surface, border, borderDim) }
            item { DashRow("COMMIT",     d.lastCommit, textPri, textSec, surface, border, borderDim) }
            item { DashRow("BRANCH",     d.branch, textPri, textSec, surface, border, borderDim) }
            item { DashRow("PENDING",    if (d.pendingWork) "YES" else "None", if (d.pendingWork) amber else green, textSec, surface, border, borderDim) }
        }
    }

    /**
     * Genesis-053 Sprint-002 â€” Approval Card
     * Rendered once per pending AWAITING_APPROVAL session in MissionControl.
     * Amber border = attention required. Green = approve. Amber = reject.
     */
    @Composable
    fun ApprovalCard(
        session:   OrchestratorSession,
        amber:     Color,
        green:     Color,
        surface:   Color,
        textPri:   Color,
        textSec:   Color,
        cyan:      Color,
        onApprove: () -> Unit,
        onReject:  () -> Unit,
    ) {
        var deciding by remember { mutableStateOf(false) }
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .background(surface, RoundedCornerShape(14.dp))
                .border(1.5.dp, amber.copy(alpha = 0.8f), RoundedCornerShape(14.dp))
                .padding(16.dp),
            verticalArrangement = Arrangement.spacedBy(10.dp)
        ) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically,
            ) {
                Text("AWAITING YOUR APPROVAL", color = amber, fontFamily = FontFamily.Monospace,
                    fontWeight = FontWeight.Bold, fontSize = 9.sp, letterSpacing = 2.sp)
                Text(session.sessionId.take(8) + "â€¦", color = textSec,
                    fontFamily = FontFamily.Monospace, fontSize = 8.sp)
            }
            Text(session.request.take(160), color = textPri, fontFamily = FontFamily.Monospace,
                fontSize = 12.sp, lineHeight = 18.sp)

            // Genesis-053 Sprint-003: plan summary
            if (session.planOperations > 0) {
                Spacer(Modifier.height(4.dp))
                Column(verticalArrangement = Arrangement.spacedBy(2.dp)) {
                    if (session.planCreates > 0)
                        Text("+ ${session.planCreates} file(s) to create", color = green,
                            fontFamily = FontFamily.Monospace, fontSize = 10.sp)
                    if (session.planModifies > 0)
                        Text("~ ${session.planModifies} file(s) to modify", color = cyan,
                            fontFamily = FontFamily.Monospace, fontSize = 10.sp)
                    if (session.planDeletes > 0)
                        Text("- ${session.planDeletes} file(s) to delete", color = amber,
                            fontFamily = FontFamily.Monospace, fontSize = 10.sp)
                    session.planFiles.take(5).forEach { file ->
                        Text("  $file", color = textSec, fontFamily = FontFamily.Monospace,
                            fontSize = 9.sp)
                    }
                }
                Spacer(Modifier.height(4.dp))
            }

            Row(horizontalArrangement = Arrangement.spacedBy(8.dp), verticalAlignment = Alignment.CenterVertically) {
                Text("STAGE", color = textSec, fontFamily = FontFamily.Monospace, fontSize = 8.sp, letterSpacing = 2.sp)
                Text(session.stage, color = amber, fontFamily = FontFamily.Monospace,
                    fontWeight = FontWeight.Bold, fontSize = 10.sp)
            }
            Spacer(Modifier.height(2.dp))
            if (deciding) {
                Row(modifier = Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.Center) {
                    CircularProgressIndicator(color = cyan, modifier = Modifier.size(24.dp), strokeWidth = 2.dp)
                }
            } else {
                Row(modifier = Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(12.dp)) {
                    Button(
                        onClick = { deciding = true; onApprove() },
                        modifier = Modifier.weight(1f),
                        colors = ButtonDefaults.buttonColors(containerColor = green.copy(alpha = 0.2f)),
                        shape = RoundedCornerShape(8.dp),
                    ) {
                        Text("APPROVE", color = green, fontFamily = FontFamily.Monospace,
                            fontWeight = FontWeight.Bold, fontSize = 11.sp, letterSpacing = 2.sp)
                    }
                    Button(
                        onClick = { deciding = true; onReject() },
                        modifier = Modifier.weight(1f),
                        colors = ButtonDefaults.buttonColors(containerColor = amber.copy(alpha = 0.15f)),
                        shape = RoundedCornerShape(8.dp),
                    ) {
                        Text("REJECT", color = amber, fontFamily = FontFamily.Monospace,
                            fontWeight = FontWeight.Bold, fontSize = 11.sp, letterSpacing = 2.sp)
                    }
                }
            }
        }
    }

    @Composable
    fun DashRow(label: String, value: String, valueColor: Color, labelColor: Color, surface: Color, border: Color, borderDim: Color) {
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .background(surface, RoundedCornerShape(10.dp))
                .border(1.dp, borderDim.copy(alpha = 0.5f), RoundedCornerShape(10.dp))
                .padding(horizontal = 14.dp, vertical = 10.dp),
            horizontalArrangement = Arrangement.SpaceBetween,
            verticalAlignment = Alignment.CenterVertically
        ) {
            Text(label, color = labelColor, fontFamily = FontFamily.Monospace, fontSize = 9.sp, letterSpacing = 2.sp)
            Text(value, color = valueColor, fontFamily = FontFamily.Monospace, fontSize = 12.sp, fontWeight = FontWeight.Bold)
        }
    }

    @Composable
    fun MissionControl(
        data: DashboardData?, cyan: Color, amber: Color, green: Color,
        surface: Color, border: Color, textPri: Color, textSec: Color, bg: Color,
        // Genesis-053 Sprint-002
        pendingSessions: List<OrchestratorSession> = emptyList(),
        onApprove: (String) -> Unit = {},
        onReject:  (String) -> Unit = {},
    ) {
        val d = data ?: DashboardData()
        LazyColumn(modifier = Modifier.fillMaxSize().padding(12.dp), verticalArrangement = Arrangement.spacedBy(10.dp)) {

            // Genesis-053 Sprint-002: all pending approval sessions at top
            if (pendingSessions.isNotEmpty()) {
                items(pendingSessions) { session ->
                    ApprovalCard(
                        session   = session,
                        amber     = amber,
                        green     = green,
                        surface   = surface,
                        textPri   = textPri,
                        textSec   = textSec,
                        cyan      = cyan,
                        onApprove = { onApprove(session.sessionId) },
                        onReject  = { onReject(session.sessionId) },
                    )
                }
            }

            item {
                Column(modifier = Modifier.fillMaxWidth()
                    .background(surface, RoundedCornerShape(14.dp))
                    .border(1.dp, border.copy(alpha = 0.5f), RoundedCornerShape(14.dp))
                    .padding(16.dp)
                ) {
                    Text("CURRENT MISSION", color = textSec, fontFamily = FontFamily.Monospace, fontSize = 9.sp, letterSpacing = 3.sp)
                    Spacer(Modifier.height(6.dp))
                    Text(d.currentMission, color = cyan, fontFamily = FontFamily.Monospace, fontWeight = FontWeight.Bold, fontSize = 15.sp)
                    Spacer(Modifier.height(14.dp))
                    Text("PROGRESS  ${d.progressPercent}%", color = textSec, fontFamily = FontFamily.Monospace, fontSize = 9.sp, letterSpacing = 2.sp)
                    Spacer(Modifier.height(6.dp))
                    Box(modifier = Modifier.fillMaxWidth().height(8.dp).background(bg, RoundedCornerShape(4.dp)).border(1.dp, border.copy(alpha = 0.3f), RoundedCornerShape(4.dp))) {
                        Box(modifier = Modifier.fillMaxWidth(d.progressPercent / 100f).fillMaxHeight().background(cyan, RoundedCornerShape(4.dp)))
                    }
                }
            }
            item {
                Column(modifier = Modifier.fillMaxWidth()
                    .background(surface, RoundedCornerShape(14.dp))
                    .border(1.dp, border.copy(alpha = 0.3f), RoundedCornerShape(14.dp))
                    .padding(16.dp)
                ) {
                    Text("TODAY'S OBJECTIVES", color = textSec, fontFamily = FontFamily.Monospace, fontSize = 9.sp, letterSpacing = 3.sp)
                    Spacer(Modifier.height(10.dp))
                    if (d.objectives.isEmpty()) {
                        Text("Loading objectives...", color = textSec, fontFamily = FontFamily.Monospace, fontSize = 12.sp)
                    } else {
                        d.objectives.forEach { obj ->
                            Row(modifier = Modifier.padding(vertical = 5.dp), verticalAlignment = Alignment.CenterVertically, horizontalArrangement = Arrangement.spacedBy(10.dp)) {
                                Text(if (obj.done) "\u2713" else "\u25a1", color = if (obj.done) green else textSec, fontFamily = FontFamily.Monospace, fontSize = 14.sp, fontWeight = FontWeight.Bold)
                                Text(obj.text, color = if (obj.done) textSec else textPri, fontFamily = FontFamily.Monospace, fontSize = 12.sp,
                                    textDecoration = if (obj.done) TextDecoration.LineThrough else null)
                            }
                        }
                    }
                }
            }
            item {
                Row(modifier = Modifier.fillMaxWidth()
                    .background(surface, RoundedCornerShape(14.dp))
                    .border(1.dp, amber.copy(alpha = 0.4f), RoundedCornerShape(14.dp))
                    .padding(16.dp),
                    horizontalArrangement = Arrangement.SpaceBetween, verticalAlignment = Alignment.CenterVertically
                ) {
                    Column {
                        Text("NEXT MILESTONE", color = textSec, fontFamily = FontFamily.Monospace, fontSize = 9.sp, letterSpacing = 3.sp)
                        Spacer(Modifier.height(4.dp))
                        Text(d.nextMilestone, color = amber, fontFamily = FontFamily.Monospace, fontWeight = FontWeight.Bold, fontSize = 13.sp)
                    }
                    Text("\u25b6", color = amber, fontFamily = FontFamily.Monospace, fontSize = 18.sp)
                }
            }
            item {
                Row(modifier = Modifier.fillMaxWidth()
                    .background(green.copy(alpha = 0.15f), RoundedCornerShape(14.dp))
                    .border(1.dp, green.copy(alpha = 0.5f), RoundedCornerShape(14.dp))
                    .padding(16.dp),
                    horizontalArrangement = Arrangement.SpaceBetween, verticalAlignment = Alignment.CenterVertically
                ) {
                    Text("STATUS", color = textSec, fontFamily = FontFamily.Monospace, fontSize = 9.sp, letterSpacing = 3.sp)
                    Text("READY", color = green, fontFamily = FontFamily.Monospace, fontWeight = FontWeight.Bold, fontSize = 13.sp, letterSpacing = 2.sp)
                }
            }
        }
    }
