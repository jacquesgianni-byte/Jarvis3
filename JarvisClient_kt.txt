package com.jarvis.android.alpha

import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.MediaType.Companion.toMediaTypeOrNull
import okhttp3.MultipartBody
import okhttp3.OkHttpClient
import okhttp3.Request
import okhttp3.RequestBody.Companion.asRequestBody
import okhttp3.RequestBody.Companion.toRequestBody
import org.json.JSONObject
import java.io.File
import java.util.concurrent.TimeUnit

/**
 * Jarvis Android Alpha
 *
 * The only network layer in the app.
 * Talks to the Jarvis Flask API. No AI, no routing, no logic.
 * One Brain. Multiple Interfaces.
 *
 * Genesis-053 Sprint-002:
 *   - serverUrl now sourced from BuildConfig.JARVIS_SERVER_URL (local.properties)
 *   - orchestratorStatus() â€” poll for AWAITING_APPROVAL sessions
 *   - approveSession()     â€” approve or reject a specific session
 */
class JarvisClient {

    // Sourced from local.properties via BuildConfig â€” never hardcoded.
    // Update JARVIS_SERVER_URL in local.properties when ngrok URL changes.
    private val serverUrl = BuildConfig.JARVIS_SERVER_URL

    // Orchestrator token â€” sourced from local.properties via BuildConfig.
    // Note: BuildConfig values end up in the APK; do not use as a truly
    // secret credential. Acceptable for private v1 development use.
    private val orchestratorToken = BuildConfig.ORCHESTRATOR_TOKEN

    private val http = OkHttpClient.Builder()
        .connectTimeout(12, TimeUnit.SECONDS)
        .readTimeout(60, TimeUnit.SECONDS)
        .writeTimeout(60, TimeUnit.SECONDS)
        .build()

    // â”€â”€ Chat â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

    suspend fun chat(message: String): Result<String> = withContext(Dispatchers.IO) {
        try {
            val body = JSONObject().apply { put("message", message) }
                .toString()
                .toRequestBody("application/json".toMediaType())

            val request = Request.Builder()
                .url("$serverUrl/chat")
                .post(body)
                .build()

            val response = http.newCall(request).execute()
            val json = JSONObject(response.body?.string() ?: "")
            Result.success(json.getString("message"))

        } catch (e: java.net.ConnectException) {
            Result.failure(Exception(
                "I can't reach your Jarvis server, sir. " +
                "Please check that it's running on your desktop."
            ))
        } catch (e: java.net.SocketTimeoutException) {
            Result.failure(Exception(
                "Your Jarvis server didn't respond in time, sir. " +
                "Please check that it's running on your desktop."
            ))
        } catch (e: Exception) {
            Result.failure(Exception("Server error, sir: ${e.message}"))
        }
    }

    // â”€â”€ File Upload â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

    suspend fun upload(
        file: File,
        mimeType: String,
        message: String = ""
    ): Result<UploadResponse> = withContext(Dispatchers.IO) {
        try {
            val fileBody = file.asRequestBody(mimeType.toMediaTypeOrNull())

            val multipart = MultipartBody.Builder()
                .setType(MultipartBody.FORM)
                .addFormDataPart("file", file.name, fileBody)
                .apply {
                    if (message.isNotBlank()) addFormDataPart("message", message)
                }
                .build()

            val request = Request.Builder()
                .url("$serverUrl/upload")
                .post(multipart)
                .build()

            val response = http.newCall(request).execute()
            val rawBody = response.body?.string() ?: "{}"
            val json = JSONObject(rawBody)

            if (!response.isSuccessful) {
                return@withContext Result.failure(
                    Exception(json.optString("message", "Upload failed."))
                )
            }

            Result.success(
                UploadResponse(
                    success      = json.optBoolean("success", false),
                    message      = json.optString("message", "File received."),
                    uploadId     = json.optString("upload_id", ""),
                    filename     = json.optString("filename", file.name),
                    detectedType = json.optString("detected_type", "unknown"),
                    readable     = json.optBoolean("readable", false),
                )
            )

        } catch (e: java.net.ConnectException) {
            Result.failure(Exception(
                "I can't reach your Jarvis server, sir. " +
                "Please check that it's running on your desktop."
            ))
        } catch (e: java.net.SocketTimeoutException) {
            Result.failure(Exception("Upload timed out, sir. The file may be too large."))
        } catch (e: Exception) {
            Result.failure(Exception("Upload error, sir: ${e.message}"))
        }
    }

    // â”€â”€ Dashboard â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

    suspend fun dashboard(): DashboardData? = withContext(Dispatchers.IO) {
        try {
            val request = Request.Builder().url("$serverUrl/dashboard").build()
            val response = http.newCall(request).execute()
            val json = org.json.JSONObject(response.body?.string() ?: "{}")
            val sys = json.optJSONObject("system") ?: org.json.JSONObject()
            val eng = json.optJSONObject("engineering") ?: org.json.JSONObject()
            DashboardData(
                status         = sys.optString("status", "online"),
                version        = sys.optString("version", "0.1-alpha"),
                uptime         = sys.optString("uptime", "--"),
                workersOnline  = sys.optInt("workers_online", 0),
                memoryStatus   = sys.optString("memory_status", "unknown"),
                memoryRecords  = sys.optInt("memory_records", 0),
                aiProvider     = sys.optString("ai_provider", "none"),
                currentGenesis = eng.optString("current_genesis", "Android Alpha"),
                lastGenesis    = eng.optString("last_completed_genesis", "Genesis-041"),
                testsPassed    = eng.optInt("tests_passed", 4110),
                lastCommit     = eng.optString("last_commit", "7f26ef0"),
                branch         = eng.optString("branch", "main"),
                pendingWork    = eng.optBoolean("pending_commit", false) || eng.optBoolean("pending_push", false)
            )
        } catch (e: Exception) { null }
    }

    // â”€â”€ Health â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

    suspend fun health(): Boolean = withContext(Dispatchers.IO) {
        try {
            val request = Request.Builder().url("$serverUrl/health").build()
            val response = http.newCall(request).execute()
            response.isSuccessful
        } catch (e: Exception) {
            false
        }
    }

    // â”€â”€ Orchestrator (Genesis-053 Sprint-002) â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

    /**
     * Poll for sessions currently AWAITING_APPROVAL.
     *
     * Returns a list of OrchestratorSession objects (may be empty).
     * Returns null on network error â€” caller should handle gracefully.
     *
     * Called every 30 seconds from the existing heartbeat in JarvisOS().
     */
    suspend fun orchestratorStatus(): List<OrchestratorSession>? = withContext(Dispatchers.IO) {
        if (orchestratorToken.isBlank()) return@withContext emptyList()
        try {
            val request = Request.Builder()
                .url("$serverUrl/orchestrator/status")
                .header("X-Orchestrator-Token", orchestratorToken)
                .get()
                .build()

            val response = http.newCall(request).execute()
            if (!response.isSuccessful) return@withContext null

            val json    = JSONObject(response.body?.string() ?: "{}")
            val arr     = json.optJSONArray("sessions") ?: return@withContext emptyList()
            val sessions = mutableListOf<OrchestratorSession>()

            for (i in 0 until arr.length()) {
                val s = arr.getJSONObject(i)
                val ps = s.optJSONObject("plan_summary")
                val planFiles = mutableListOf<String>()
                ps?.optJSONArray("files")?.let { arr ->
                    for (i in 0 until arr.length()) planFiles.add(arr.optString(i))
                }
                sessions.add(
                    OrchestratorSession(
                        sessionId      = s.optString("session_id", ""),
                        request        = s.optString("request", ""),
                        stage          = s.optString("stage", "AWAITING_APPROVAL"),
                        status         = s.optString("status", "AWAITING_APPROVAL"),
                        approvedBy     = s.optString("approved_by", ""),
                        approvedAt     = s.optString("approved_at", ""),
                        planOperations = ps?.optInt("operations", 0) ?: 0,
                        planCreates    = ps?.optInt("creates", 0) ?: 0,
                        planModifies   = ps?.optInt("modifies", 0) ?: 0,
                        planDeletes    = ps?.optInt("deletes", 0) ?: 0,
                        planFiles      = planFiles,
                    )
                )
            }
            sessions
        } catch (e: Exception) {
            null   // network error â€” caller shows no change
        }
    }

    /**
     * Approve or reject a specific session.
     *
     * @param sessionId  The session UUID from OrchestratorSession.sessionId
     * @param decision   "approve" or "reject"
     * @param reason     Required for rejection; ignored for approval
     *
     * Returns true if the server accepted the decision.
     * Returns false on error or if session is no longer pending (already processed).
     */
    suspend fun approveSession(
        sessionId: String,
        decision:  String,
        reason:    String = "Rejected via Android app",
    ): Boolean = withContext(Dispatchers.IO) {
        try {
            val body = JSONObject().apply {
                put("session_id",  sessionId)
                put("decision",    decision)
                put("decided_by",  "ludovic")
                put("reason",      reason)
            }.toString().toRequestBody("application/json".toMediaType())

            val request = Request.Builder()
                .url("$serverUrl/orchestrator/approve")
                .header("X-Orchestrator-Token", orchestratorToken)
                .post(body)
                .build()

            val response = http.newCall(request).execute()
            // 404 = already processed (idempotent) â€” treat as success, not error
            if (response.code == 404) return@withContext true
            response.isSuccessful
        } catch (e: Exception) {
            false
        }
    }
}

// â”€â”€ Data classes â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

data class UploadResponse(
    val success:      Boolean,
    val message:      String,
    val uploadId:     String,
    val filename:     String,
    val detectedType: String,
    val readable:     Boolean,
)

/**
 * A single session currently awaiting Ludovic's approval.
 * Populated from GET /orchestrator/status response.
 * Genesis-053 Sprint-002.
 */
data class OrchestratorSession(
    val sessionId:    String,
    val request:      String,   // truncated request description
    val stage:        String,
    val status:       String,
    val approvedBy:   String,
    val approvedAt:   String,
    // Genesis-053 Sprint-003: plan summary
    val planOperations: Int     = 0,
    val planCreates:    Int     = 0,
    val planModifies:   Int     = 0,
    val planDeletes:    Int     = 0,
    val planFiles:      List<String> = emptyList(),
)
